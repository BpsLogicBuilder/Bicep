$rg = "rg-testbicep"
az deployment group create --resource-group $rg --template-file acr/acr.bicep

//Look up $acrName from the portal
$acrName="bpscrqbdozxe5lmnsc"
//Derive $aksCluster using the same suffix
$aksCluster="bps-aks-qbdozxe5lmnsc"

//From repository root folder
az acr build --registry $acrName --image webapplication1:v1.0.0 --image webapplication1:latest ./WebApplication1

az deployment group create --resource-group $rg --template-file kubernetes/k8s.bicep --parameters dnsPrefix="$($aksCluster)" linuxAdminUsername=azureuser sshRSAPublicKey=''

//Assign AcrPull permissions for the ACR
az deployment group create --resource-group $rg --template-file kubernetes/assign-acr-pull-to-aks.bicep

//Get AKS credentials
az aks get-credentials --resource-group $rg --name $aksCluster --overwrite-existing

//Image deployment
#sed -e "s|ACR_NAME|${acr_name}|g" kubernetes/deployment.yaml | kubectl apply -f - -n default
$deploymentContent = Get-Content kubernetes/deployment.yaml -Raw
$deploymentContent = $deploymentContent -replace "ACR_NAME", $acrName
$deploymentContent | kubectl apply -f - -n default

kubectl apply -f kubernetes/deployment.yaml -n default
